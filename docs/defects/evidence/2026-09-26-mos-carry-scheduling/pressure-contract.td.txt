def C : Register<"c">;
def V : Register<"v">;

// that set even when the first class with those units permits pruning.
def ACc : RegisterClass<"Test", [i1], 8, (add C)>;
def ZCc : RegisterClass<"Test", [i8], 8, (add C)> {
  let isPressureFineGrained = true;
}
// An ordinary single-register subset may still merge with its superset.
def Vc : RegisterClass<"Test", [i8], 8, (add V)>;
def CVc : RegisterClass<"Test", [i8], 8, (add C, V)>;
def Test : Target;
