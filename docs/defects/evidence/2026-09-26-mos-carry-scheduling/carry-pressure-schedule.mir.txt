# Finish the scaling shift before starting the independent add carry. Keeping
# both flags live would require saving one in a byte register.
---
name: scale_then_add
tracksRegLiveness: true
body: |
  bb.0:
    %0:ac = LDAbs 1000
    %1:ac = LDAbs 1001
    %2:ac = LDAbs 1002
    %3:ac = LDAbs 1003
    %4:cc = LDImm1 0
    %0:ac, %5:cc = ASL %0
    %1:ac, %5:cc = ROL %1, %5
    %2:ac, %5:cc = ROL %2, %5
    %3:ac, dead %5:cc = ROL %3, %5
    %0:ac, %6:cc, dead %10:vc = ADCImm %0, 23, %4
    %1:ac, %7:cc, dead %11:vc = ADCImm %1, 45, %6
    %2:ac, %8:cc, dead %12:vc = ADCImm %2, 67, %7
    %3:ac, dead %9:cc, dead %13:vc = ADCImm %3, 0, %8
    STAbs %0, 2000
    STAbs %1, 2001
    STAbs %2, 2002
    STAbs %3, 2003
    RTS
...

# CHECK-LABEL: name: scale_then_add
# CHECK: ASL
# CHECK-NOT: ADC
# CHECK: dead {{%[0-9]+}}:cc = ROL
# CHECK: ADCImm
