from pathlib import Path
import subprocess,json,re,hashlib
root=Path.cwd();out=root/'build/carry-gate/interpreter';out.mkdir(exist_ok=True);tool=root/'build/carry-gate/candidate-install/bin';commands=[];rows=[]
def run(cmd,label):
 cmd=list(map(str,cmd));p=subprocess.run(cmd,capture_output=True,text=True,timeout=300);(out/(label+'.log')).write_text(p.stdout+p.stderr);commands.append({'command':cmd,'exit_code':p.returncode});assert p.returncode==0,p.stderr;return p.stdout
for policy in ('off','always','gated'):
 obj=root/'build/carry-gate/focused'/('examples_snes_corpus_lsystem_sim.c.xy16.Oz.'+policy+'.o');rom=out/(policy+'.sfc')
 run([tool/'mos-clang','--config',root/'.scratch/carry-scheduling/build/install/bin/mos-snes.cfg','-mcpu=mosw65816','-fno-lto',obj,'-o',rom],policy+'-link')
 run(['python3',root/'tools/snes-checksum.py',rom],policy+'-checksum')
 symbols={p[-1]:(int(p[0],16),int(p[-2],16)) for line in run([tool/'llvm-objdump','-t',str(rom)+'.elf'],policy+'-symbols').splitlines() if len(p:=line.split())>=5 and re.fullmatch('[0-9a-f]{8}',p[0])}
 main,size=symbols['main'];offset,length=symbols['corpus_result'];data=rom.read_bytes();body=data[main-0x8000:main-0x8000+size]
 # The interpreter initializes heading 192 after the L-system generation loop.
 marker=bytes.fromhex('e220a2c08a');assert body.count(marker)==1
 start=main+body.index(marker)
 run([tool/'llvm-objdump','-d',str(rom)+'.elf'],policy+'-disassembly')
 cmd=[root/'.scratch/carry-timing/jgxcycles',rom,root/'vendor/bsnes-jg/Database',hex(start),hex(main+size),'0xffffffff',hex(offset),length,'0x8073',2000,1]
 runs=[json.loads(run(cmd,policy+'-timing-'+str(i))) for i in range(2)];assert runs[0]==runs[1] and runs[0]['pass']
 row={'policy':policy,'start_offset':start-main,'marker':marker.hex(),'rom_sha256':hashlib.sha256(data).hexdigest(),'samples':runs[0]['samples'],'repeat_identical':True,'command':list(map(str,cmd))};rows.append(row);print(json.dumps(row),flush=True)
(out/'results.json').write_text(json.dumps({'runs':rows,'commands':commands},indent=2)+'\n')
