// Get the number of dimensions of register pressure.
unsigned MOSGenRegisterInfo::getNumRegPressureSets() const {
  return 6;
}

// Get the name of this register unit pressure set.
const char *MOSGenRegisterInfo::
getRegPressureSetName(unsigned Idx) const {
  static const char *PressureNameTable[] = {
    "MOSAsmParamRegClass",
    "Pc",
    "GPR_LSB",
    "GPR_LSB_with_Pc",
    "Anyi1",
    "Anyi1_with_Pc",
  };
  return PressureNameTable[Idx];
}

// Get the register unit pressure limit for this dimension.
// This limit must be adjusted dynamically for reserved registers.
unsigned MOSGenRegisterInfo::
getRegPressureSetLimit(const MachineFunction &MF, unsigned Idx) const {
  static const uint16_t PressureLimitTable[] = {
    3,  	// 0: MOSAsmParamRegClass
    4,  	// 1: Pc
    6,  	// 2: GPR_LSB
    7,  	// 3: GPR_LSB_with_Pc
    262,  	// 4: Anyi1
    263,  	// 5: Anyi1_with_Pc
  };
  return PressureLimitTable[Idx];
}

/// Get the register class for this pressure set with the largest
/// `RegClassWeight::WeightLimit`.
const TargetRegisterClass *MOSGenRegisterInfo::
getLargestRegClassForRegPressureSet(unsigned Idx) const {
  static const uint8_t PSetRegClassTable[] = {
    MOS::MOSAsmParamRegClassRegClassID,  	// 0: MOSAsmParamRegClass
    MOS::PcRegClassID,  	// 1: Pc
    MOS::CV_GPR_LSBRegClassID,  	// 2: GPR_LSB
    MOS::CV_GPR_LSBRegClassID,  	// 3: GPR_LSB_with_Pc
    MOS::Anyi1RegClassID,  	// 4: Anyi1
    MOS::Anyi1RegClassID,  	// 5: Anyi1_with_Pc
  };
  return getRegClass(PSetRegClassTable[Idx]);
}
