# RUN: llc -mtriple=mos -mcpu=mosw65816 -run-pass=scavenger-test -verify-machineinstrs %s -o - | FileCheck %s
#
# The scavenger harness consumes block-local scratch virtual registers directly.
# The first virtual value ends before X and Y become live. Both index registers
# and carry are live across the later temporary carry value. Extending its
# status save to the earlier virtual-register use would cross PH $a without
# its matching PL. Keep PHP/PLP inside the balanced accumulator push/pull pair.
# The final ADC consumes the original carry; the stored bytes are 6, 6, 7, 1.
#
# CHECK-LABEL: name: status_save_range
# CHECK: bb.1:
# CHECK: $x = LDImm 6
# CHECK-NEXT: $y = LDImm 7
# CHECK-NEXT: PH $a
# CHECK-NEXT: PH $p
# CHECK-NEXT: $c = LDImm1 0
# CHECK-NEXT: $a, $c, $v = ADCImm $a, 0, $c
# CHECK-NEXT: $p = PL
# CHECK-NEXT: $a = PL
# CHECK-NEXT: $a, dead $c, dead $v = ADCImm $a, 0, $c
# CHECK-NEXT: STAbs $a, @scavenged_values

--- |
  target triple = "mos"
  @scavenged_values = external global [4 x i8]
  define void @status_save_range() { unreachable }
...
---
name: status_save_range
tracksRegLiveness: true
body: |
  bb.0:
    successors: %bb.1
    $a = LDImm 5
    $c = LDImm1 1
    BRA %bb.1

  bb.1:
    liveins: $a, $c

    %0:gpr = LDImm 1
    STAbs %0, @scavenged_values + 3
    $x = LDImm 6
    $y = LDImm 7
    PH $a
    undef %1.subcarry:pc = LDImm1 0
    $a, dead %1.subcarry, dead %1.subv = ADCImm $a, 0, %1.subcarry
    $a = PL
    $a, dead $c, dead $v = ADCImm $a, 0, $c
    STAbs $a, @scavenged_values
    STAbs $x, @scavenged_values + 1
    STAbs $y, @scavenged_values + 2
    RTS
...
