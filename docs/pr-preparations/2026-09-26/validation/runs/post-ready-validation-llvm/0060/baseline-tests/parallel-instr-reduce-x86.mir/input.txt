# REQUIRES: thread_support, x86-registered-target
# RUN: llvm-reduce -x=mir -j 1 --delta-passes=instructions --test FileCheck --test-arg --check-prefix=INTERESTING --test-arg %s --test-arg --input-file %s -o %t.serial
# RUN: llvm-reduce -x=mir -j 2 --delta-passes=instructions --test FileCheck --test-arg --check-prefix=INTERESTING --test-arg %s --test-arg --input-file %s -o %t.parallel
# RUN: FileCheck --check-prefix=CHECK %s < %t.parallel
# RUN: diff %t.serial %t.parallel

# MIR reductions retain machine functions when more than one worker is requested.
# Both job counts use the same serial reduction order and preserve the witness.
# INTERESTING: MOV32ri 2
# CHECK: MOV32ri 2
# CHECK: RET

--- |
  target triple = "x86_64"
  define i32 @parallel_instr_reduce() { ret i32 0 }
...
---
name: parallel_instr_reduce
tracksRegLiveness: true
body: |
  bb.0:
    %0:gr32 = MOV32ri 1
    %1:gr32 = MOV32ri 2
    %2:gr32 = ADD32rr %0, %1, implicit-def dead $eflags
    $eax = COPY %2
    RET 0, implicit $eax
...
